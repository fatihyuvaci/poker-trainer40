  function initNavigation() {

  function selectPos(posKey) {
    if (!NAV_40BB.spots[posKey]) return;
    currentPosKey = posKey;
    document.querySelectorAll('.pos-btn').forEach(b => b.classList.remove('active'));
    const posBtn = document.getElementById(`posbtn-${posKey}`);
    if (posBtn) {
      posBtn.classList.add('active');
      posBtn.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
    }

    const spots = NAV_40BB.spots[posKey];
    const spotContainer = document.getElementById('spotButtonsContainer');
    spotContainer.innerHTML = spots.map(s => `
      <button class="spot-btn" id="spotbtn-${s.id}" onclick="selectSpot('${s.id}')">${s.label}</button>
    `).join('');

    selectSpot(spots[0].id);
  }

  function selectSpot(spotId) {
    selectedSpotId = spotId;
    document.querySelectorAll('.spot-btn').forEach(b => b.classList.remove('active'));
    const spBtn = document.getElementById(`spotbtn-${spotId}`);
    if (spBtn) {
      spBtn.classList.add('active');
      spBtn.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
    }
    dealNextHand();
  }

  function resolveScenarioFromSpot(spotId) {
    const pick = arr => SCENARIOS[arr[Math.floor(Math.random() * arr.length)]];
    if (spotId === 'mix_40_utg') {
      const utgKeys = Object.keys(SCENARIOS).filter(k => SCENARIOS[k].group === 'UTG');
      return pick(utgKeys);
    }
    if (spotId === 'mix_40_hj') {
      const hjKeys = Object.keys(SCENARIOS).filter(k => SCENARIOS[k].group === 'HJ');
      return pick(hjKeys);
    }
    if (spotId === 'mix_40_bn') {
      const bnKeys = Object.keys(SCENARIOS).filter(k => SCENARIOS[k].group === 'BN');
      return pick(bnKeys);
    }
    if (spotId === 'mix_40_sb') {
      const sbKeys = Object.keys(SCENARIOS).filter(k => SCENARIOS[k].group === 'SB');
      return pick(sbKeys);
    }
    if (spotId === 'mix_40_bb') {
      const bbKeys = Object.keys(SCENARIOS).filter(k => SCENARIOS[k].group === 'BB');
      return pick(bbKeys);
    }
    if (spotId === 'mix_40_all') {
      return pick(Object.keys(SCENARIOS));
    }
    return SCENARIOS[spotId];
  }

  function getHandBreakdown(scenario, hand) {
    const entry = scenario.raw[hand];
    if (!entry) {
      if (scenario.isConditional) return null;
      return { allin: 0, raise: 0, call: 0, fold: 100 };
    }
    const allin = entry.allin || 0;
    const raise = entry.raise || 0;
    const call  = entry.call  || 0;
    const used = allin + raise + call;
    const fold = entry.fold !== undefined ? entry.fold : Math.max(0, 100 - used);
    return { allin, raise, call, fold };
  }

  function buildScenarioUI(scenario) {
    document.getElementById('badgePos').textContent = scenario.pos;
    document.getElementById('badgeStack').textContent = `${scenario.stack} BB`;
    document.getElementById('badgeSpot').textContent = scenario.spot;
    document.getElementById('feltBanner').textContent = scenario.banner;
    document.getElementById('chartTitle').textContent = scenario.title;

    const legEl = document.getElementById('chartLegend');
    legEl.innerHTML = scenario.legend.map(item => `
      <div class="legend-item">
        <div class="legend-swatch" style="background:${item.color};"></div>
        <span>${item.label}</span>
      </div>
    `).join('');

    const btnContainer = document.getElementById('actionButtonsContainer');
    btnContainer.innerHTML = scenario.actions.map(act => `
      <button class="act-btn ${act.cls}" id="actbtn-${act.code}" onclick="submitAction('${act.code}')">
        <span class="act-title">${act.label}</span>
        <span class="act-sub">${act.sub}</span>
      </button>
    `).join('');

    if (lastRenderedScenarioId === scenario.id) {
      document.querySelectorAll('.matrix-cell.active-hand').forEach(el => el.classList.remove('active-hand'));
      return;
    }
    lastRenderedScenarioId = scenario.id;

    const grid = document.getElementById('matrixGrid');
    grid.innerHTML = '';

    ALL_169_HANDS.forEach(hand => {
      const cell = document.createElement('div');
      cell.id = `cell-${hand}`;
      const dist = getHandBreakdown(scenario, hand);

      if (!dist) {
        cell.className = 'matrix-cell unused-cell';
        cell.innerHTML = `<span>${hand}</span>`;
      } else {
        cell.className = 'matrix-cell';
        const bars = document.createElement('div');
        bars.className = 'cell-bars';

        if (dist.allin > 0) {
          const b = document.createElement('div');
          b.className = 'bar-allin';
          b.style.width = `${dist.allin}%`;
          bars.appendChild(b);
        }
        if (dist.raise > 0) {
          const b = document.createElement('div');
          b.className = 'bar-raise';
          b.style.width = `${dist.raise}%`;
          bars.appendChild(b);
        }
        if (dist.call > 0) {
          const b = document.createElement('div');
          b.className = 'bar-call';
          b.style.width = `${dist.call}%`;
          bars.appendChild(b);
        }
        if (dist.fold > 0) {
          const b = document.createElement('div');
          b.className = 'bar-fold';
          b.style.width = `${dist.fold}%`;
          bars.appendChild(b);
        }

        cell.appendChild(bars);
        const label = document.createElement('span');
        label.textContent = hand;
        cell.appendChild(label);
      }
      grid.appendChild(cell);
    });
  }

  function pickHandNotation(scenario) {
    const mode = document.getElementById('drillMode').value;
    const eligibleHands = scenario.isConditional
      ? Object.keys(scenario.raw)
      : ALL_169_HANDS;

    if (mode === 'borderline') {
      return scenario.borderline[Math.floor(Math.random() * scenario.borderline.length)];
    } else if (mode === 'smart') {
      if (Math.random() < 0.65) {
        return scenario.borderline[Math.floor(Math.random() * scenario.borderline.length)];
      }
    }
    return eligibleHands[Math.floor(Math.random() * eligibleHands.length)];
  }

  function renderPlayingCard(elId, rank, suitObj) {
    const card = document.getElementById(elId);
    card.className = `playing-card ${suitObj.cls}`;
    card.innerHTML = `
      <div class="corner-rank">${rank}</div>
      <div class="center-suit">${suitObj.symbol}</div>
      <div class="bottom-rank">${rank}</div>
    `;
  }

  function handleFeltTap() {
    if (answered) {
      dealNextHand();
    }
  }

  function dealNextHand() {
    answered = false;
    isHoldingBackquote = false;
    maskWasCoveredBeforeHold = true;
    activeScenario = resolveScenarioFromSpot(selectedSpotId);

    buildScenarioUI(activeScenario);
    document.getElementById('matrixMask').classList.remove('hidden');
    document.getElementById('pokerFelt').classList.remove('clickable-felt');

    const notation = pickHandNotation(activeScenario);
    currentHand = notation;

    const r1 = notation[0];
    const r2 = notation[1];
    const idx1 = Math.floor(Math.random() * 4);
    const idx2 = (idx1 + 1 + Math.floor(Math.random() * 3)) % 4;
    const s1 = SUITS[idx1];
    const s2 = (notation.length === 3 && notation[2] === 's') ? s1 : SUITS[idx2];

    renderPlayingCard('card1', r1, s1);
    renderPlayingCard('card2', r2, s2);
    document.getElementById('handNotation').textContent = notation;

    const fbBox = document.getElementById('feedbackBox');
    fbBox.className = 'feedback-box';
    const fbIcon = document.getElementById('feedbackIcon');
    fbIcon.className = 'feedback-icon icon-neutral';
    fbIcon.textContent = '?';
    document.getElementById('feedbackTitle').textContent = `${activeScenario.pos} (40bb • ${activeScenario.spot})`;
    document.getElementById('feedbackDesc').textContent = 'Select your action below.';
    document.getElementById('nextBtn').style.display = 'none';
  }

  function formatBreakdownText(dist) {
    const parts = [];
    let callLabel = 'Call';
    if (activeScenario.id === 'sb_pfi') callLabel = 'Limp';
    else if (activeScenario.id === 'bb_vs_sb_limp') callLabel = 'Check';

    if (dist.allin > 0) parts.push(`All-in: ${dist.allin}%`);
    if (dist.raise > 0) parts.push(`Raise/3B/4B: ${dist.raise}%`);
    if (dist.call > 0)  parts.push(`${callLabel}: ${dist.call}%`);
    if (dist.fold > 0)  parts.push(`Fold: ${dist.fold}%`);
    return parts.join(' • ');
  }

  function submitAction(actionCode) {
    if (answered) return;
    answered = true;
    maskWasCoveredBeforeHold = false;

    const dist = getHandBreakdown(activeScenario, currentHand);
    const chosenFreq = dist[actionCode] || 0;
    const isCorrect = chosenFreq > 0;

    try {
      if (navigator.vibrate) {
        navigator.vibrate(isCorrect ? 25 : [50, 40, 50]);
      }
    } catch (e) {}

    stats.total++;
    if (isCorrect) {
      stats.correct++;
      stats.streak++;
    } else {
      stats.streak = 0;
    }
    saveStats();

    const fbBox = document.getElementById('feedbackBox');
    const fbIcon = document.getElementById('feedbackIcon');
    const fbTitle = document.getElementById('feedbackTitle');
    const fbDesc = document.getElementById('feedbackDesc');

    const breakdownStr = formatBreakdownText(dist);

    if (isCorrect) {
      fbBox.className = 'feedback-box correct';
      fbIcon.className = 'feedback-icon icon-correct';
      fbIcon.textContent = '✓';
      fbTitle.textContent = chosenFreq === 100
        ? `CORRECT! Pure 100% (${currentHand})`
        : `CORRECT! Mixed (${chosenFreq}%) — ${currentHand}`;
    } else {
      fbBox.className = 'feedback-box wrong';
      fbIcon.className = 'feedback-icon icon-wrong';
      fbIcon.textContent = '✗';
      fbTitle.textContent = `INCORRECT! (${currentHand})`;
    }

    fbDesc.textContent = `GTO: [ ${breakdownStr} ]`;
    document.getElementById('nextBtn').style.display = 'block';
    document.getElementById('pokerFelt').classList.add('clickable-felt');

    document.getElementById('matrixMask').classList.add('hidden');
    const activeCell = document.getElementById(`cell-${currentHand}`);
    if (activeCell) activeCell.classList.add('active-hand');

    document.querySelectorAll('.act-btn').forEach(b => b.disabled = true);
  }

  function togglePeek() {
    const mask = document.getElementById('matrixMask');
    mask.classList.toggle('hidden');
    maskWasCoveredBeforeHold = !mask.classList.contains('hidden');
  }

  function loadStats() {
    try {
      const saved = localStorage.getItem('nlh40bb_23spot_stats');
      if (saved) stats = JSON.parse(saved);
    } catch (e) {}
    updateStatsUI();
  }

  function saveStats() {
    try {
      localStorage.setItem('nlh40bb_23spot_stats', JSON.stringify(stats));
    } catch (e) {}
    updateStatsUI();
  }

  function resetStats() {
    stats = { correct: 0, total: 0, streak: 0 };
    saveStats();
  }

  function updateStatsUI() {
    document.getElementById('scoreText').textContent = `${stats.correct}/${stats.total}`;
    const acc = stats.total > 0 ? Math.round((stats.correct / stats.total) * 100) : 0;
    document.getElementById('accText').textContent = `${acc}%`;
    document.getElementById('streakText').textContent = stats.streak;
  }

  window.addEventListener('keydown', (e) => {
    if (e.code === 'Backquote' || e.key === '`') {
      e.preventDefault();
      if (!isHoldingBackquote) {
        isHoldingBackquote = true;
        const mask = document.getElementById('matrixMask');
        maskWasCoveredBeforeHold = !mask.classList.contains('hidden');
        mask.classList.add('hidden');
      }
      return;
    }

    const key = e.key.toLowerCase();
    if (!answered) {
      const acts = activeScenario.actions;
      if (key === '1' && acts[0]) submitAction(acts[0].code);
      else if (key === '2' && acts[1]) submitAction(acts[1].code);
      else if (key === '3' && acts[2]) submitAction(acts[2].code);
      else if (key === '4' && acts[3]) submitAction(acts[3].code);
      else if (key === 'f' && acts.some(a => a.code === 'fold')) submitAction('fold');
      else if (key === 'c' && acts.some(a => a.code === 'call')) submitAction('call');
      else if (key === 'r' && acts.some(a => a.code === 'raise')) submitAction('raise');
      else if (key === 'a' && acts.some(a => a.code === 'allin')) submitAction('allin');
    } else {
      if (key === ' ' || key === 'enter') {
        e.preventDefault();
        dealNextHand();
      }
    }
  });

  window.addEventListener('keyup', (e) => {
    if (e.code === 'Backquote' || e.key === '`') {
      e.preventDefault();
      if (isHoldingBackquote) {
        isHoldingBackquote = false;
        const mask = document.getElementById('matrixMask');
        if (maskWasCoveredBeforeHold) {
          mask.classList.remove('hidden');
        }
      }
    }
  });

  loadStats();
  initNavigation();
